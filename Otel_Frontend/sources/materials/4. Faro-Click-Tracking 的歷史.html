<pre><code>import { BaseInstrumentation } from '@grafana/faro-web-sdk';

const CLICK_EVENT_NAME = 'click';
const THROTTLE_MS = 300;

export interface ClickInstrumentationOptions {
  /**
   * 選填: 要額外追蹤的 data-* attribute 名稱清單。每次點擊時,各自從 event.target
   * 開始以 closest() 往上查找最近一個帶有該 attribute 的元素(含自身),找到則將其值
   * 併入同一筆 click event payload; 找不到則該欄位不出現在 payload 中,不視為錯誤。
   *
   * payload 的 key 不會直接採用原始 attribute 名稱: 經由 api.pushEvent() 送到 Alloy 後,
   * 欄位名稱會統一加上 event_data_ 前綴,若原始 attribute 名稱帶有連字號(例如 data-* 慣例的
   * data-panel-topic),組出的欄位名稱(event_data_data-panel-topic)會因為包含連字號
   * 而無法被 Loki query 解析。因此送進 payload 前,會先用 toPayloadKey 轉換: 去除
   * data- 前綴、其餘連字號改為底線,例如 data-panel-topic -> panel_topic,
   * 對應 Loki 上的 event_data_panel_topic。
   *
   * payload 完全、僅由此清單命中的欄位組成,套件不會 default 帶入任何欄位
   * (例如過去的 link_name/device_type/max_touch_points)。
   */
  trackAttributes?: string[];
}

// 將 data-* attribute 名稱轉換為 click event payload 使用的 key: 去除 "data-" 前綴,
// 其餘連字號改為底線。Loki 端會在 payload key 前自動加上 event_data_ 前綴,
// 若 key 本身仍帶連字號(例如未轉換的 data-panel-topic),組出的
// event_data_data-panel-topic 欄位名稱無法被 Loki query 解析,因此統一轉換為底線格式。
export function toPayloadKey(attributeName: string): string {
  return attributeName.replace(/^data-/, '').replace(/-/g, '_');
}

// 對單一 attribute 名稱,從 element 開始(含自身)往上查找最近一個帶有該 attribute 的元素,
// 回傳其原始屬性值(未 trim); 找不到則回傳 undefined,交由呼叫端視為「這次點擊沒有這個欄位」,
// 不當作錯誤處理。
function getTrackedAttributeValue(element: Element, attributeName: string): string | undefined {
  const marked = element.closest(`[${attributeName}]`);
  if (marked === null) {
    return undefined;
  }
  return marked.getAttribute(attributeName) ?? undefined;
}

export class ClickInstrumentation extends BaseInstrumentation {
  readonly name = 'faro-click-tracking:instrumentation-click';
  readonly version = '0.1.0';

  private readonly trackAttributes: string[];

  // 同一個 target 300ms 內只送一次,用 WeakMap 避免持有 DOM 節點造成記憶體洩漏。
  private lastSentAt = new WeakMap<EventTarget, number>();

  constructor(options: ClickInstrumentationOptions = {}) {
    super();
    this.trackAttributes = options.trackAttributes ?? [];
  }

  private handleClick = (event: MouseEvent): void => {
    const target = event.target;
    if (!(target instanceof Element)) {
      return;
    }

    // payload 完全由 trackAttributes 查找結果組成,套件不 default 帶入任何欄位;
    // 各自獨立查找,找到才併入 payload,找不到就省略該欄位,不視為錯誤。
    // payload 的 key 統一轉換過(見 toPayloadKey()),而非直接使用原始 attribute 名稱。
    const payload: Record<string, string> = {};
    for (const attributeName of this.trackAttributes) {
      const value = getTrackedAttributeValue(target, attributeName);
      if (value !== undefined) {
        payload[toPayloadKey(attributeName)] = value;
      }
    }

    // 沒有任何 trackAttributes 命中時,這次點擊沒有可送的資訊,不送出事件(也不佔用節流時間戳記)。
    if (Object.keys(payload).length === 0) {
      return;
    }

    const now = Date.now();
    const lastSent = this.lastSentAt.get(target);
    if (lastSent !== undefined && now - lastSent < THROTTLE_MS) {
      return;
    }
    this.lastSentAt.set(target, now);

    this.api.pushEvent(CLICK_EVENT_NAME, payload);
  };

  initialize(): void {
    document.addEventListener('click', this.handleClick);
  }

  destroy(): void {
    document.removeEventListener('click', this.handleClick);
  }

import type { FaroUserInfo } from './faroUser';
import { clearFaroUser, setFaroUser } from './faroUser';

/**
 * 讀取「目前使用者身分」的宿主端函式。
 * 必須是同步、便宜的操作(會在 click 等使用者互動當下被呼叫),
 * 未登入時回傳 null。套件不假設、也不需要知道實際的儲存位置與格式。
 */
export type GetUserFn = () => FaroUserInfo | null;

/**
 * 依 getUser() 回傳結果自動同步身分至 Faro 的 instrumentation。
 * 與 ClickInstrumentation 職責分離: 本身不做任何點擊事件追蹤,只負責身分同步,
 * 複用 Faro BaseInstrumentation 的 lifecycle 管理三個觸發同步時機的 listener。
 */
export class UserSyncInstrumentation extends BaseInstrumentation {
  readonly name = 'faro-click-tracking:instrumentation-user-sync';
  readonly version = '0.1.0';

  private readonly getUser: GetUserFn;
  // 未定義代表「尚未同步過」; null 視為「已知目前未登入」,兩者都與有值的 id 不同,故合併以 undefined 表示。
  private lastKnownId: string | undefined;
  // 套件不假設宿主的建置工具(Vite/webpack/esbuild 等),無法可靠判斷 production 與否,
  // 因此僅在「第一次」發生例外時提示一次,避免每次觸發都洗版,而非以環境變數判斷。
  private hasWarnedGetUserError = false;

  constructor(getUser: GetUserFn) {
    super();
    this.getUser = getUser;
  }

  // 供 initFaro() 於掛載 instrumentations 之前,立即執行一次初始同步(處理頁面載入時已登入的情境)。
  sync = (): void => {
    let user: FaroUserInfo | null;
    try {
      user = this.getUser();
    } catch (error) {
      // getUser() 由宿主專案自行實作,品質不受套件控制; 不應讓例外中斷使用者當下的互動(例如 click)。
      if (!this.hasWarnedGetUserError) {
        this.hasWarnedGetUserError = true;
        console.warn('[faro-click-tracking] getUser() 執行時拋出例外,本次視為未登入:', error);
      }
      user = null;
    }

    const nextId = user?.id;
    if (nextId === this.lastKnownId) {
      return;
    }
    this.lastKnownId = nextId;

    if (user) {
      setFaroUser(user);
    } else {
      clearFaroUser();
    }
  };

  initialize(): void {
    document.addEventListener('click', this.sync);
    document.addEventListener('visibilitychange', this.sync);
    window.addEventListener('storage', this.sync);
  }

  destroy(): void {
    document.removeEventListener('click', this.sync);
    document.removeEventListener('visibilitychange', this.sync);
    window.removeEventListener('storage', this.sync);
  }
}</code></pre>

/**
 * 套件內建的 environment → collector domain 對照表。
 *
 * key 為完整的 environment 字串(例如 'tw-stage'),不做任何切分/解析 ——
 * environment 的命名慣例(地區前綴、段數)由使用方自行決定(例如未來可能出現 'tw-sre-test' 這種三段式值),

 *
 * th 地區相關項目目前為佔位值(TODO-*),待實際 Alloy domain 確認後於此處直接更新即可,
 * 不需要調整查表機制本身。
 */
const COLLECTOR_DOMAIN_BY_ENVIRONMENT: Record<string, string> = {
  'tw-test': 'https://pek8s-testing.garmin.com',
  'tw-stage': 'https://pek8s-staging.garmin.com',
  'tw-prod': 'https://shixpa-peproxy00.garmin.com',
  'tw-uat': 'https://pek8s-uat.garmin.com',
  'tw-exp': 'https://xinxta-pisvc00.garmin.com',
  'th-test': 'https://pek8s-th-testing.garmin.com',
};

/**
 * 依 environment 於套件內建對照表查找 collector domain,並拼接固定 path suffix 組成完整 Alloy Faro receiver URL。
 *
 * 套件不再接受宿主專案自行提供 collector domain: 同一個團隊每個部署環境的 Alloy domain 是固定值,
 * 由套件統一管理,宿主專案只需提供 environment 即可。

 * @returns 完整的 Alloy Faro receiver URL
 * @throws 若 environment 查無對應的內建 collector domain,拋出 Error 並附上目前傳入的值
 */
export function resolveCollectorUrl(environment: string): string {
  const domain = COLLECTOR_DOMAIN_BY_ENVIRONMENT[environment];
  if (!domain) {
    const known = Object.keys(COLLECTOR_DOMAIN_BY_ENVIRONMENT).join(', ');
    throw new Error(
      `[faro-click-tracking] 找不到 environment "${environment}" 對應的 collector domain。` +
        `目前支援的 environment 值: ${known}。`,
    );
  }
  return `${domain}${ALLOY_PATH_SUFFIX}`;

// 套件目前設計為模組級單例: 一個網頁應用程式對應 initFaro() 建立的一個 Faro 實例。
// 供套件內任何非 instrumentation 的對外 API(不像 ClickInstrumentation 能由 Faro SDK
// 框架自動注入 api)在需要直接存取 Faro 實例時使用。
let faroInstance: Faro | undefined;

function assertNotInitialized(): void {
  if (faroInstance !== undefined) {
    throw new Error(
      '[faro-click-tracking] initFaro() 已被呼叫過,不可重複呼叫; 請確認應用程式生命週期中只呼叫一次。',
    );
  }
}

/**
 * 內部使用: 供 initFaro() 在建立新的 Faro 實例(呼叫 initializeFaro())之前先行檢查,
 * 避免重複呼叫 initFaro() 時仍執行到一半才失敗(已產生副作用,例如已對 collector 送出訊號)。
 * 尚未呼叫過 initFaro() 時為 no-op; 已呼叫過時拋出明確錯誤。
 */
export function ensureNotInitialized(): void {
  assertNotInitialized();
}

/**
 * 內部使用: 由 initFaro() 呼叫,保存本次初始化建立的 Faro 實例參照。
 * 不對外匯出,宿主專案不需要、也不應該直接取得 Faro 實例。
 *
 * @throws 若已註冊過 Faro 實例(即 initFaro() 已被呼叫過),拋出 Error 阻止重複註冊。
 */
export function registerFaroInstance(faro: Faro): void {
  assertNotInitialized();
  faroInstance = faro;
}

/**
 * 內部使用: 取得已註冊的 Faro 實例; 若尚未呼叫過 initFaro(),拋出 Error 提示需先初始化。
 * callerName 用於在錯誤訊息中標示是哪個對外 API 呼叫失敗,方便使用者定位問題。
 */
export function getFaroInstanceOrThrow(callerName: string): Faro {
  if (!faroInstance) {
    throw new Error(
      `[faro-click-tracking] 尚未呼叫 initFaro(),請先初始化再呼叫 ${callerName}()。`,
    );
  }
  return faroInstance;
